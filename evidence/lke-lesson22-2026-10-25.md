## Repeatable verification

```bash
KCFG="$HOME/.kube/twn-capstone04-lke.yaml"
NS=online-shop
kubectl --kubeconfig "$KCFG" get nodes
kubectl --kubeconfig "$KCFG" -n "$NS" get deployments,pods,services
for resource in $(kubectl --kubeconfig "$KCFG" -n "$NS" get deployments -o name); do
  kubectl --kubeconfig "$KCFG" -n "$NS" rollout status "$resource" --timeout=180s
done
kubectl --kubeconfig "$KCFG" -n "$NS" port-forward --address 127.0.0.1 service/frontend 18080:80
# In second terminal: curl --noproxy '*' -sS -o /dev/null -w 'HTTP %{http_code}\n' http://127.0.0.1:18080/
```
